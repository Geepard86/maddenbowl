Reset/Entziehen funktioniert nicht? Einmalig diese SQL-Funktionen in Supabase anlegen (SQL Editor) 
create or replace function public.mb_reset_achievements(pw_hash text)
returns json language plpgsql security definer set search_path = public as $$
declare a int; p int; f int;
begin
  if pw_hash is null or pw_hash <> (select value from app_config where key = 'admin_pw_hash') then
    raise exception 'admin check failed';
  end if;
  delete from player_achievements where true; get diagnostics a = row_count;
  delete from player_progress where true;     get diagnostics p = row_count;
  delete from achievement_firsts where true;  get diagnostics f = row_count;
  return json_build_object('player_achievements', a, 'player_progress', p, 'achievement_firsts', f);
end $$;
grant execute on function public.mb_reset_achievements(text) to anon, authenticated;

-- Für „Karte entziehen“:
create or replace function public.mb_revoke_achievement(pw_hash text, p_player text, p_id text)
returns json language plpgsql security definer set search_path = public as $$
declare a int; f int;
begin
  if pw_hash is null or pw_hash <> (select value from app_config where key = 'admin_pw_hash') then
    raise exception 'admin check failed';
  end if;
  delete from player_achievements where lower(player_name) = lower(p_player) and achievement_id = p_id; get diagnostics a = row_count;
  delete from achievement_firsts where lower(player_name) = lower(p_player) and achievement_id = p_id; get diagnostics f = row_count;
  return json_build_object('removed', a, 'unique_freed', f);
end $$;
grant execute on function public.mb_revoke_achievement(text, text, text) to anon, authenticated;
